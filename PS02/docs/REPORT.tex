\section{fork() แบบมีและไม่มี wait()}
% =====================================================================

\subsection{Source code}

\subsubsection{1.1 โปรแกรม helloworld (ไม่มี \code{wait()})}

\begin{lstlisting}[caption={q1\_1\_helloworld.c -- fork() without wait()},label={lst:q1_1}]
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main(void)
{
    printf("Hello World!\n");

    pid_t pid = fork();
    if (pid < 0) {
        perror("fork");
        exit(EXIT_FAILURE);
    }

    if (pid == 0) {
        /* child */
        printf("I am the child, my PID = %d, my parent PID = %d\n",
               getpid(), getppid());
    } else {
        /* parent */
        printf("I am the parent, my PID = %d, my child PID = %d\n",
               getpid(), pid);
    }

    printf("Common line printed by PID %d\n", getpid());
    return 0;
}
\end{lstlisting}

\textbf{คำอธิบายการทำงาน}: เรียก \code{fork()} เพื่อสร้าง child process หากเป็น child (\code{pid == 0}) จะพิมพ์ PID ของตนเองและ parent ส่วน parent จะพิมพ์ PID ของตนเองและลูก เนื่องจากไม่มี \code{wait()} parent จึงจบการทำงานโดยไม่รอ child

\newpage

\subsubsection{1.2 โปรแกรมที่เพิ่ม \code{wait()}}

\begin{lstlisting}[caption={q1\_2\_wait.c -- fork() with wait()},label={lst:q1_2}]
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main(void)
{
    printf("Hello World!\n");

    pid_t pid = fork();
    if (pid < 0) {
        perror("fork");
        exit(EXIT_FAILURE);
    }

    if (pid == 0) {
        /* child */
        printf("I am the child, my PID = %d, my parent PID = %d\n",
               getpid(), getppid());
    } else {
        /* parent */
        wait(NULL);
        printf("I am the parent, my PID = %d, my child PID = %d\n",
               getpid(), pid);
    }

    printf("Common line printed by PID %d\n", getpid());
    return 0;
}
\end{lstlisting}

\textbf{คำอธิบายการทำงาน}: เพิ่มการเรียก \code{wait(NULL)} ในฝั่ง parent เพื่อสั่งให้ parent หยุดรอ (Blocked) จนกว่า child จะทำงานเสร็จและคืนสถานะเรียบร้อย บังคับให้ลำดับการแสดงผลของ child เกิดขึ้นก่อน parent เสมอ

\subsection{อภิปรายผล}

จากการทดลองเปรียบเทียบระหว่างโปรแกรมข้อ 1.1 (ไม่มี \code{wait()}) และข้อ 1.2 (มี \code{wait()}) สรุปความแตกต่างที่สำคัญได้ดังนี้

\begin{itemize}[leftmargin=*]
  \item \textbf{การซิงโครไนซ์และลำดับ Output}:
        ในข้อ 1.1 ทั้ง Parent และ Child แย่งกันทำงานอย่างอิสระ ลำดับการพิมพ์ข้อความจึงขึ้นอยู่กับ Kernel Scheduler และไม่แน่นอน (Non-deterministic) ในขณะที่ข้อ 1.2 การเรียก \code{wait(NULL)} จะทำให้ Parent หยุดรอ (Blocked) จนกว่า Child จะทำงานเสร็จ ลำดับ Output จึงมีความแน่นอน (Deterministic) โดยข้อความของ Child จะปรากฏก่อนเสมอ

  \item \textbf{ปรากฏการณ์ Orphan Process}:
        ในข้อ 1.1 มีโอกาสที่ Parent จะทำงานเสร็จและจบโปรแกรมไปก่อน ทำให้ Child กลายเป็น Orphan Process และถูกเคอร์เนลยกให้ \code{init} (หรือ Subreaper) เป็น Parent ใหม่แทน (สังเกตได้จาก PPID ของ Child ที่เปลี่ยนไปในการรันครั้งที่ 2) ส่วนในข้อ 1.2 จะไม่เกิดเหตุการณ์นี้เนื่องจาก Parent รอ Child อยู่เสมอ
\end{itemize}

\section{Process Status และ sleep()}
% =====================================================================

การทดลองนี้ใช้ \code{sleep()} เป็นเครื่องมือควบคุม \emph{ลำดับการตาย} ของ parent
และ child เพื่อสร้างสถานการณ์สองแบบที่ตรงข้ามกัน แล้วใช้คำสั่ง \code{ps}
ตรวจสอบสถานะของ process ในช่วงเวลาที่สนใจ ทั้งสองโปรแกรม \textbf{จงใจไม่เรียก}
\code{wait()} เพื่อให้เห็นพฤติกรรมที่เกิดขึ้นเมื่อ parent ไม่เก็บกวาดลูกของตนเอง

\subsection{Source code}

\subsubsection{2.1 Child ตายก่อน โดย Parent ยังทำงานอยู่}

\begin{lstlisting}[caption={q2\_1\_child\_first.c -- child dies first, parent never reaps},label={lst:q2_1}]
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

#define PARENT_LIFETIME 30   /* seconds the parent stays alive after child dies */

int main(void)
{
    pid_t pid = fork();

    if (pid < 0) {
        perror("fork");
        exit(EXIT_FAILURE);
    }

    if (pid == 0) {
        /* CHILD: dies almost immediately */
        printf("[child ] PID = %d, PPID = %d -- exiting right now\n",
               getpid(), getppid());
        fflush(stdout);
        exit(0);
    }

    /* PARENT: outlives the child but never reaps it */
    printf("[parent] PID = %d, child PID = %d\n", getpid(), pid);
    printf("[parent] NOT calling wait(). Sleeping %d s.\n", PARENT_LIFETIME);
    printf("[parent] Inspect now with:  ps -ef | grep -e q2_1 -e defunct\n");
    fflush(stdout);

    sleep(PARENT_LIFETIME);

    printf("[parent] PID = %d exiting; the zombie child %d is now re-parented "
           "to init/systemd and reaped.\n", getpid(), pid);
    return 0;
}
\end{lstlisting}

\textbf{คำอธิบายการทำงาน}: หลัง \code{fork()} ฝั่ง child จะพิมพ์ PID ของตนเองแล้วเรียก \code{exit(0)} ทันที ส่วน parent เข้าสู่ \code{sleep(30)} ทำให้เกิดช่วงเวลา 30 วินาทีที่ child ตายไปแล้วแต่ parent ยังมีชีวิตอยู่และยังไม่ได้เรียก \code{wait()} จึงเป็นช่วงที่เปิดอีก terminal หนึ่งเข้าไปตรวจด้วย \code{ps} ได้ทัน การเรียก \code{fflush(stdout)} ก่อน \code{exit()} และก่อน \code{sleep()} เป็นการบังคับให้ข้อความถูกเขียนออกทันที ไม่ค้างอยู่ใน buffer

\newpage

\subsubsection{2.2 Parent ตายก่อน โดย Child ยังทำงานอยู่}

\begin{lstlisting}[caption={q2\_2\_parent\_first.c -- parent dies first, child becomes an orphan},label={lst:q2_2}]
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

#define CHILD_LIFETIME 30

int main(void)
{
    pid_t pid = fork();

    if (pid < 0) {
        perror("fork");
        exit(EXIT_FAILURE);
    }

    if (pid == 0) {
        /* CHILD: outlives the parent */
        printf("[child ] PID = %d, PPID = %d  (original parent, still alive)\n",
               getpid(), getppid());
        fflush(stdout);

        sleep(2);   /* give the parent time to die */

        printf("[child ] PID = %d, PPID = %d  (parent is gone -> re-parented)\n",
               getpid(), getppid());
        printf("[child ] Inspect now with:  ps -ef | grep q2_2\n");
        fflush(stdout);

        sleep(CHILD_LIFETIME);

        printf("[child ] PID = %d exiting, final PPID = %d\n",
               getpid(), getppid());
        exit(0);
    }

    /* PARENT: exits immediately, without waiting */
    printf("[parent] PID = %d, child PID = %d -- exiting immediately\n",
           getpid(), pid);
    fflush(stdout);
    return 0;
}
\end{lstlisting}

\textbf{คำอธิบายการทำงาน}: สลับบทบาทจากข้อ 2.1 คือ parent พิมพ์ข้อความแล้วจบการทำงานทันที ส่วน child อยู่ต่ออีก 30 วินาที จุดสำคัญคือ child เรียก \code{getppid()} \textbf{สองครั้ง} โดยมี \code{sleep(2)} คั่นกลาง ครั้งแรกพิมพ์ตอน parent ยังมีชีวิต ครั้งที่สองพิมพ์หลัง parent ตายไปแล้ว ทำให้เห็นค่า PPID ที่เปลี่ยนไปได้จากตัวโปรแกรมเองโดยไม่ต้องพึ่ง \code{ps}

\subsection{อภิปรายผล}

จากการทดลองควบคุมลำดับการจบการทำงานของกระบวนการด้วย \code{sleep()} ในข้อ 2.1 และ 2.2 สรุปข้อแตกต่างและประเด็นสำคัญได้ดังนี้

\begin{itemize}[leftmargin=*]
  \item \textbf{ข้อ 2.1 ปรากฏการณ์ Zombie Process}:
        เมื่อ Child ตายก่อนแต่ Parent ยังทำงานอยู่โดยไม่ได้เรียก \code{wait()} เคอร์เนลจะคืนหน่วยความจำของ Child ทันที (\code{SZ = 0}) แต่ยังคงจอง PID และ Process Table Entry ค้างไว้ ทำให้เกิดสถานะ \code{Z} หรือ \code{<defunct>} จนกว่า Parent จะจบโปรแกรมหรือเรียก \code{wait()} เพื่ออ่าน Exit Status

  \item \textbf{ข้อ 2.2 ปรากฏการณ์ Orphan Process และการ Reparent}:
        เมื่อ Parent ตายก่อนโดยที่ Child ยังคงทำงานอยู่ Child จะกลายเป็นกระบวนการกำพร้า (Orphan) เคอร์เนลจะยก Child ให้กับ Subreaper ที่ใกล้ที่สุด (เช่น \code{systemd} หรือ \code{/init} ของ WSL) เป็น Parent คนใหม่ทันที สังเกตได้จากค่า PPID ของ Child ที่เปลี่ยนไปในการเรียก \code{getppid()} ครั้งที่สอง

  \item \textbf{ความแตกต่างสำคัญและปรากฏการณ์ \code{<defunct>}}:
        Zombie (\code{<defunct>}) คือกระบวนการที่สิ้นสุดการทำงานแล้วแต่ยังถูกจองค้างในตารางกระบวนการ หากสะสมมากจะสิ้นเปลือง PID และทำให้สร้าง Process ใหม่ไม่ได้ ต่างจาก Orphan ที่เป็นกระบวนการซึ่งยังคงทำงานตามปกติ และเมื่อ Orphan สิ้นสุดการทำงาน Parent คนใหม่ (Subreaper) จะทำหน้าที่เรียก \code{wait()} เก็บกวาดให้อัตโนมัติโดยไม่เกิด Zombie ทิ้งไว้
\end{itemize}

\section{จำนวน process สูงสุดที่ fork ได้}
% =====================================================================

การทดลองนี้ต้องการหาว่าระบบยอมให้เรียก \code{fork()} ได้สำเร็จกี่ครั้ง
โดยใช้โครงสร้างแบบ \textbf{โซ่ (chain)} ตามที่ Hint ของโจทย์แนะนำ คือทุก process
สร้างลูกเพียงตัวเดียวแล้วรอด้วย \code{waitpid()} โซ่จึงยาวขึ้นทีละหนึ่งและคลายตัว
กลับจากปลายสุดอย่างเป็นระเบียบ ซึ่ง \textbf{ไม่ใช่ fork bomb} และหยุดได้ทุกเมื่อ

\subsection{Source code}

\subsubsection{ส่วนหลัก: ฟังก์ชัน \code{chain()} และการรายงานผล}

\begin{lstlisting}[caption={q3\_forkcount.c -- the chain and the reporting path},label={lst:q3a}]
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/resource.h>

#define HARD_CAP 200000        /* never build a chain longer than this */

struct result {
    int count;                 /* number of successful fork() calls */
    int err;                   /* errno that stopped us (0 = hit HARD_CAP) */
};

static int pfd[2];

/* Send the tally up the chain to the root and leave. */
static _Noreturn void report(int count, int err)
{
    struct result r;
    r.count = count;
    r.err   = err;
    if (write(pfd[1], &r, sizeof r) != (ssize_t)sizeof r)
        _exit(1);
    _exit(0);
}

#pragma GCC diagnostic push
#pragma GCC diagnostic ignored "-Winfinite-recursion"

/* I am the process produced by fork() call number `level`. */
static void chain(int level)
{
    pid_t pid;

    if (level >= HARD_CAP)
        report(level, 0);      /* stopped by our own ceiling */

    pid = fork();

    if (pid < 0)
        /* fork #(level+1) failed -> `level` forks succeeded in total */
        report(level, errno);

    if (pid == 0)
        chain(level + 1);      /* the child keeps digging */

    /* parent of this link: hold still until the chain below me unwinds */
    waitpid(pid, NULL, 0);
    _exit(0);
}

#pragma GCC diagnostic pop
\end{lstlisting}

\textbf{คำอธิบายการทำงาน}: \code{chain(level)} หมายถึง ``ฉันคือ process ที่เกิดจาก \code{fork()} ครั้งที่ \code{level}'' แต่ละตัวจะ \code{fork()} หนึ่งครั้ง ถ้าสำเร็จ ลูกจะเรียก \code{chain(level+1)} ขุดลึกต่อไป ส่วนพ่อจะติดอยู่ที่ \code{waitpid()} ไม่ทำอะไรจนกว่าโซ่ข้างล่างจะคลายกลับมา ทำให้มีเพียง process เดียวเท่านั้นที่ทำงานอยู่ในแต่ละขณะ เมื่อ \code{fork()} ล้มเหลว process ที่ลึกที่สุดจะส่งผลรวมและ \code{errno} กลับไปให้ root ผ่าน pipe ด้วย \code{report()} ส่วน \code{HARD\_CAP} เป็นเพดานของโปรแกรมเองไว้กันพลาด และคำสั่ง \code{\#pragma} จำเป็นต้องมีเพราะ gcc มองว่า \code{chain()} เป็น infinite recursion ซึ่งเป็นการเตือนที่ผิดพลาด เนื่องจากการเรียกซ้ำเกิดใน process ใหม่คนละตัวและหยุดเมื่อ \code{fork()} ล้มเหลว

\newpage

\subsubsection{ส่วน \code{main()}: กำหนดเพดานและรับผลลัพธ์}

\begin{lstlisting}[caption={q3\_forkcount.c -- main(): optional ceiling and result collection},label={lst:q3b}]
int main(int argc, char *argv[])
{
    pid_t pid;
    struct result r = { 0, 0 };
    struct rlimit rl;

    if (argc > 1) {            /* optional artificial ceiling for rehearsal */
        rlim_t want = (rlim_t)strtoul(argv[1], NULL, 10);
        if (getrlimit(RLIMIT_NPROC, &rl) == 0) {
            rl.rlim_cur = want;
            if (rl.rlim_max != RLIM_INFINITY && rl.rlim_cur > rl.rlim_max)
                rl.rlim_cur = rl.rlim_max;
            if (setrlimit(RLIMIT_NPROC, &rl) != 0)
                perror("setrlimit");
        }
    }

    if (getrlimit(RLIMIT_NPROC, &rl) == 0)
        printf("RLIMIT_NPROC in effect: soft = %ld, hard = %ld\n",
               (long)rl.rlim_cur, (long)rl.rlim_max);

    printf("Root PID = %d. Forking a chain until fork() fails ...\n", getpid());
    fflush(stdout);

    if (pipe(pfd) < 0) {
        perror("pipe");
        exit(EXIT_FAILURE);
    }

    pid = fork();
    if (pid < 0) {
        r.count = 0;
        r.err   = errno;
    } else if (pid == 0) {
        close(pfd[0]);
        chain(1);              /* this process came from fork() #1 */
        _exit(0);              /* unreachable */
    } else {
        close(pfd[1]);         /* only the descendants write */
        waitpid(pid, NULL, 0);
        if (read(pfd[0], &r, sizeof r) != (ssize_t)sizeof r) {
            fprintf(stderr, "could not read result from pipe\n");
            exit(EXIT_FAILURE);
        }
        close(pfd[0]);
    }

    printf("fork() succeeded %d times.\n", r.count);
    if (r.err == 0)
        printf("Stopped by the program's own HARD_CAP (%d), not by the OS.\n",
               HARD_CAP);
    else
        printf("fork() #%d failed with errno %d (%s).\n",
               r.count + 1, r.err, strerror(r.err));

    return 0;
}
\end{lstlisting}

\textbf{คำอธิบายการทำงาน}: \code{main()} สร้าง pipe ก่อนแล้วจึง \code{fork()} เพื่อเริ่มโซ่ ฝั่ง root ปิดปลายเขียน (\code{pfd[1]}) เพราะมีแต่ลูกหลานที่จะเขียน ส่วนลูกตัวแรกปิดปลายอ่าน (\code{pfd[0]}) การที่ root เรียก \code{waitpid()} ก่อนแล้วค่อย \code{read()} ไม่ทำให้เกิด deadlock เพราะข้อมูลขนาด 8 ไบต์ถูกเขียนค้างไว้ใน buffer ของ pipe ตั้งแต่ตอนที่ process ปลายสุดยังมีชีวิตอยู่ และเนื่องจากขนาดน้อยกว่า \code{PIPE\_BUF} การเขียนจึงเป็น atomic ไม่มีทางถูกแทรก ส่วนอาร์กิวเมนต์ \code{argv[1]} เป็นทางเลือกให้ตั้งเพดานจำลองด้วย \code{setrlimit()} เพื่อซ้อมการทดลองในระดับที่ปลอดภัยก่อนรันจริง

\subsection{อภิปรายผล}

จากการทดลองวัดจำนวน \code{fork()} ที่สำเร็จภายใต้สภาพระบบที่ต่างกัน
สรุปประเด็นสำคัญได้ดังนี้

\begin{itemize}[leftmargin=*]
  \item \textbf{ขีดจำกัดที่แท้จริงคือ \code{RLIMIT\_NPROC}}:
        \code{fork()} return $-1$ พร้อม \code{errno} $= 11$ ซึ่งคือ \code{EAGAIN}
        (Resource temporarily unavailable) เหมือนกันทุกครั้งในทุกการทดลอง
        ไม่ใช่ \code{ENOMEM} แสดงว่าสิ่งที่หยุดเราคือโควตาจำนวน process ต่อผู้ใช้
        ไม่ใช่หน่วยความจำหรือขนาดของตารางกระบวนการ

  \item \textbf{จำนวนที่ได้ไม่เท่ากับเพดานเพราะโควตาถูกใช้ไปก่อนแล้ว}:
        เพดาน 300 ได้ 255 ครั้ง (ส่วนต่าง 45) และเพดาน 6000 ได้ 5953 ครั้ง
        (ส่วนต่าง 47) เนื่องจาก \code{RLIMIT\_NPROC} นับ process ของ UID นั้น
        ทั้ง user namespace ไม่ใช่นับเฉพาะที่โปรแกรมเราสร้าง ส่วนต่างจึงคือ
        process ที่ผู้ใช้มีอยู่ก่อนแล้ว เช่น shell, WSL relay และ session อื่น
        ที่เปิดค้างไว้ สังเกตว่า \code{ps} มองเห็นเพียง 8 ตัวใน session ปัจจุบัน
        แต่ถูกหักไป 47 แสดงว่ามีอีกราว 39 ตัวที่อยู่นอก session ปัจจุบัน
        แต่ยังถูกนับรวมในโควตาเดียวกัน

  \item \textbf{จำนวนลดลงตามจำนวนโปรแกรมที่รันค้างแบบหนึ่งต่อหนึ่ง}:
        เมื่อเปิดโปรแกรม \code{sleep} ค้างไว้ 100 ตัว จำนวน \code{fork()}
        ที่สำเร็จลดลงจาก 5953 เหลือ 5853 คือลดลง \textbf{100 พอดีเป๊ะ}
        ไม่ขาดไม่เกิน เช่นเดียวกับชุด A ที่เปิด 20 ตัวแล้วลดลง 20 พอดี
        ยืนยันว่าโควตานี้เป็นทรัพยากรที่แชร์กันทั้งผู้ใช้ ไม่ใช่โควตาต่อโปรแกรม

  \item \textbf{ค่าจะกลับมาเท่าเดิมก็ต่อเมื่อควบคุมตัวแปรได้ครบ}:
        ชุด B กลับมาที่ 5953 เท่าเดิมเป๊ะ เพราะปิดเฉพาะ PID ที่เปิดเองและตรวจนับ
        thread ยืนยันว่ากลับมาที่ 8 เท่าเดิมก่อนวัดซ้ำ ในขณะที่ชุด A ได้ 256 ครั้ง
        ซึ่งมากกว่าเดิม 1 เพราะระหว่างการทดลองมี process อื่นของผู้ใช้จบไปพอดี
        ทำให้โควตาว่างเพิ่มขึ้น จึงสรุปได้ว่าตัวเลขที่วัดได้เป็นค่าที่ขึ้นกับสภาพ
        ของระบบ ณ ขณะนั้น ไม่ใช่ค่าคงที่ประจำเครื่อง

  \item \textbf{ต้นทุนของโครงสร้างแบบ recursive chain}:
        การวัดแต่ละครั้งในชุด B ใช้เวลา 755--895 วินาที และใช้หน่วยความจำสูงสุด
        ราว 13 GB ที่ความยาวโซ่ประมาณ 5{,}900 process เพราะ process ที่ระดับ $N$
        มี stack frame สะสม $N$ เฟรม ทุกครั้งที่ \code{fork()} เคอร์เนลต้องคัดลอก
        page table ที่ใหญ่ขึ้นเรื่อย ๆ ทำให้ทั้งเวลาและหน่วยความจำโตแบบ $O(N^2)$
        นี่คือเหตุผลที่ไม่สามารถรันจนถึงเพดานจริง 62{,}827 ได้บนเครื่องที่มี
        หน่วยความจำ 15.7 GB และเป็นเหตุผลที่ต้องกำหนด \code{ulimit -u}
        ให้ต่ำลงก่อนการทดลองทุกครั้ง
\end{itemize}

\section*{สรุปผลการทดลอง}
\addcontentsline{toc}{section}{สรุปผลการทดลอง}
% =====================================================================

\textbf{สิ่งที่ได้เรียนรู้จากข้อ 1}

\begin{enumerate}[leftmargin=*]
  \item \code{fork()} คือการเรียกฟังก์ชันเพียงครั้งเดียวที่ return สองครั้ง
        เป็นกลไกพื้นฐานที่ทำให้โค้ดชุดเดียวแตกออกเป็นสองเส้นทางการทำงาน
  \item ถ้าไม่มีกลไกซิงโครไนซ์ ลำดับการทำงานของ parent กับ child เป็นสิ่งที่
        \textbf{คาดเดาไม่ได้} และขึ้นกับ scheduler ล้วน ๆ การรันโปรแกรมเดิมซ้ำสามครั้ง
        ให้ผลไม่ซ้ำกันเลย
  \item \code{wait()} เปลี่ยนโปรแกรมจาก non-deterministic เป็น deterministic
        และเป็นวิธีป้องกัน zombie process ที่ตรงไปตรงมาที่สุด
  \item ได้เห็น \textbf{orphan process} และการ reparent โดยบังเอิญจากการรันข้อ 1.1
        ซึ่งเป็นปรากฏการณ์เดียวกับที่ข้อ 2.2 ต้องการให้ทดลอง
  \item \code{fork()} คัดลอก \textbf{ทุกอย่าง} ใน address space รวมถึง stdio buffer
        ที่ยังไม่ได้ flush ทำให้เกิดข้อความซ้ำเมื่อ redirect output ลงไฟล์
        เป็นกับดักที่พบได้บ่อยและแก้ได้ด้วย \code{fflush(stdout)} ก่อน \code{fork()}
\end{enumerate}

\textbf{ปัญหาที่พบและวิธีแก้}

\begin{itemize}[leftmargin=*]
  \item \emph{ข้อความ \code{"Hello World!"} ปรากฏสองครั้ง} เมื่อ redirect output ลงไฟล์
        ตอนแรกเข้าใจผิดว่าโปรแกรมเขียนผิด ภายหลังพบว่าเกิดจาก stdio buffering
        แก้โดยเพิ่ม \code{fflush(stdout);} ก่อนเรียก \code{fork()}
  \item \emph{ผลการรันไม่ซ้ำเดิม} ทำให้บันทึกผลได้ยาก จึงต้องรันหลายครั้งและรายงานทุกครั้ง
        เพื่อแสดงให้เห็นความไม่แน่นอน แทนที่จะรายงานเพียงครั้งเดียว
\end{itemize}
