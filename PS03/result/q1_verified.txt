----- 3.3 process ที่ดักจับ SIGTERM ไว้ : kill เฉย ๆ ไม่พอ ต้องใช้ kill -9 -----
[sigdemo] pid = 5794, catching SIGTERM(15) and SIGINT(2)
$ kill 5794            # SIGTERM -> the handler runs, the process survives
[sigdemo] caught SIGTERM (15) -- handler ran, still alive
$ kill -INT 5794       # SIGINT  -> the handler runs, the process survives
[sigdemo] caught SIGINT  (2)  -- handler ran, still alive
$ ps -o pid,stat,comm -p 5794
    PID STAT COMMAND
   5794 S+   q1_sigdemo
$ kill -9 5794         # SIGKILL can be neither caught nor ignored

----- 3.4 kill -STOP / kill -CONT : พักและปลุก process แทนการฆ่า -----
$ ps -o pid,stat,comm -p 5803        # S = sleeping
    PID STAT COMMAND
   5803 S+   sleep
$ kill -STOP 5803
$ ps -o pid,stat,comm -p 5803        # T = stopped
    PID STAT COMMAND
   5803 T+   sleep
$ kill -CONT 5803
$ ps -o pid,stat,comm -p 5803        # back to S again
    PID STAT COMMAND
   5803 S+   sleep

----- 3.5 การฆ่า JOB ด้วย job spec %n -----
$ jobs -l
[1]   5813 Running                 sleep 400 &
[2]-  5814 Running                 sleep 500 &
[3]+  5815 Running                 sleep 600 &
$ kill %2            # send SIGTERM to job number 2
[2]-  Terminated              sleep 500
$ jobs -l
[1]-  5813 Running                 sleep 400 &
[3]+  5815 Running                 sleep 600 &
$ kill -9 %1 %3      # kill the two remaining jobs
bash:  5813 Killed                  sleep 400
bash:  5815 Killed                  sleep 600
$ jobs -l            # no output = no job left
